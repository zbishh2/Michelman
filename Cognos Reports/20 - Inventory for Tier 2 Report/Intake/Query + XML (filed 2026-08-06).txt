select 
       ITEM.BRANCH_PLANT  as  Branch_Plant,
       INVENTORY_ON_HAND_MEASURE.ITEM_NUMBER_2ND  as  C_2nd_Item_Number,
       ITEM.BULK_ITEM  as  Bulk_Item,
       ITEM.GLOBAL_BULK_ITEM  as  Global_Bulk_Item,
       INVENTORY_ON_HAND.MASTER_PLANNING_FAMILY  as  Master_Planning_Family,
       XSUM((INVENTORY_ON_HAND_MEASURE.QUANTITY_ON_HAND * INVENTORY_ON_HAND_MEASURE.CONVERSION_FACTOR_LB)  for ITEM.BULK_ITEM,ITEM.GLOBAL_BULK_ITEM,ITEM.BRANCH_PLANT,INVENTORY_ON_HAND.MASTER_PLANNING_FAMILY,INVENTORY_ON_HAND.INVENTORY_DATE,INVENTORY_ON_HAND_MEASURE.ITEM_NUMBER_2ND )  as  Quantity_on_Hand_LBs,
       INVENTORY_ON_HAND.INVENTORY_DATE  as  Inventory_Date
 from 
       "Data Warehouse"..DW_LEGACY.ITEM ITEM,
       "Data Warehouse"..DW_LEGACY.INVENTORY_ON_HAND_MEASURES INVENTORY_ON_HAND_MEASURE,
       "Data Warehouse"..DW_LEGACY.INVENTORY_ON_HAND INVENTORY_ON_HAND
 where 
       (INVENTORY_ON_HAND.INVENTORY_DATE between TIMESTAMP '2020-01-01 00:00:00.000' and TIMESTAMP '2020-12-31 00:00:00.000') and 
       (ITEM.BRANCH_PLANT in (N'CINC',N'CIN2')) and 
       (INVENTORY_ON_HAND_MEASURE.INVENTORY_ON_HAND__ITEM_SID = ITEM.ITEM_SID) and 
       (INVENTORY_ON_HAND.INVENTORY_ON_HAND_SID = INVENTORY_ON_HAND_MEASURE.INVENTORY_ON_HAND_SID)
 group by 
       ITEM.BRANCH_PLANT,
       INVENTORY_ON_HAND_MEASURE.ITEM_NUMBER_2ND,
       ITEM.BULK_ITEM,
       ITEM.GLOBAL_BULK_ITEM,
       INVENTORY_ON_HAND.MASTER_PLANNING_FAMILY,
       INVENTORY_ON_HAND.INVENTORY_DATE
