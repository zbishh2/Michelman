/* ------------------------------------------------------------------
   Report : 19 - 1 - Inventory - Safety Stock and Order Size
   Query  : Planner Names         (hidden lookup, about 57 rows)
   Source : ODSPROD / ODS / PRODDTA (JDE), SQL Server
   Note   : Native T-SQL as executed by Power BI.

            This is the ONLY query in report 19 that runs against ODS;
            the other two run against EDWPROD / EDW. It exists because
            the Cognos "Planner Name" column renders 'Last, First' in
            ASCII, while EDW's BIQL.TbItemBranch.[Planner Name] renders
            'First Last' and keeps non-ASCII accents - they differ on
            all 177 rows of the Safety Stock sheet. F0101.ABALPH is the
            string Cognos prints.

            It is a separate query rather than a join because EDWPROD
            and ODSPROD are different servers. The join is made in the
            Power BI model instead: a many-to-one relationship on
            [Planner Number], resolved by a DAX calculated column.
   ------------------------------------------------------------------ */

SET NOCOUNT ON;

SELECT
    ab.ABAN8                    AS [Planner Number],
    LTRIM(RTRIM(ab.ABALPH))     AS [Planner Name (JDE)]
FROM PRODDTA.F0101 ab WITH (NOLOCK)
WHERE ab.ABAN8 IN (
          SELECT ibp.IBANPL
          FROM PRODDTA.F4102 ibp WITH (NOLOCK)
          WHERE LTRIM(RTRIM(ibp.IBMCU)) IN ('CINC','CIN2','AUBA','AUB2','SING','SNG4')
      )
