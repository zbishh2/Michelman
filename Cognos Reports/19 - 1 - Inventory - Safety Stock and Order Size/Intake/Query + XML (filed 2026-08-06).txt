select distinct 
       ITEM.BRANCH_PLANT  as  Branch_Plant,
       ITEM.BULK_ITEM  as  Bulk_Item,
       ITEM.ITEM_NUMBER_2ND  as  C_2nd_Item_Number,
       ITEM.STOCK_TYPE_CODE  as  Stock_Type_Code,
       ITEM.MASTER_PLANNING_FAMILY__IMPR  as  Master_Planning_Family,
       ITEM.LEADTIME_MFG  as  Lead_Time_Order_to_Ship,
       ITEM.PLANNER_NUMBER  as  Planner_Number,
       VENDOR_ALIAS_PLANNER.VENDOR_NAME  as  Planner_Name,
       ITEM.SAFETY_STOCK  as  Safety_Stock,
       ITEM.UNIT_OF_MEASURE__PRIMARY  as  Unit_of_Measure_Primary
 from 
       "Data Warehouse"..DW_LEGACY.ITEM ITEM,
       "Data Warehouse"..DW_LEGACY.VENDOR VENDOR_ALIAS_PLANNER
 where 
       (ITEM.BRANCH_PLANT in (N'CINC',N'CIN2',N'AUBA',N'AUB2',N'SING',N'SNG4')) and 
       (ITEM.SAFETY_STOCK > 1) and 
       (not (ITEM.STOCK_TYPE_CODE in (N'O'))) and 
       (ITEM.MASTER_PLANNING_FAMILY__IMPR like N'%F%') and 
       (ITEM.PLANNER_NUMBER = VENDOR_ALIAS_PLANNER.VENDOR_DIM_ID)
 order by 
       Bulk_Item asc,
       C_2nd_Item_Number asc,
       Branch_Plant asc
